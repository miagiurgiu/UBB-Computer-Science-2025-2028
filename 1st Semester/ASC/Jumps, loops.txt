3. Given a string of bytes s, copy into d only the even numbers, keeping their original order.

jpe , jpo

bits 32
global start
extern exit
import exit msvcrt.dll

	s db 2,3,7,8,10
	l equ ($-s)
	d times l db 0
segment code use32 class=code:
	mov esi, 0
	mov edi, 0
	mov ecx, l
	jecxz end_loop
	start_loop:
		mov al, [s+esi]
		test al, 1
		jpo skip
			mov [d+edi], al
			inc edi
		skip:
		inc esi
	loop start_loop
	end_loop:

6. teacher's lab example 1:

; se dau 2 siruri de caractere s1 si s2.
; sa se construiasca sirul D ce contine toate elementele din s1 care nu apar in s2
bits 32
global start
extern exit
import exit msvcrt.dll
segment data use32 class=data
    s1 db '+', '4', '2', 'a', '8', '4', 'X', '5'
    l1 equ $-s1
    s2 db 'a', '4', '5'
    l2 equ $-s2
    D times l1 db 0
    
segment code use32 class=code
    start:
        mov ecx, l1; punem lungimea in ecx pt a putea realiza bucla loop de ecx ori
        mov esi, 0
        mov edi, 0
        jecxz Sfarsit
        Repeta:
            mov al, [s1+esi] ; in al avem informatia curenta din S1
            ; parcurgem sirul S2 si verificam 
            mov edx, 1 ; 1 daca nu apare in s2, 0 daca apare
            mov ebx, 0 ; index in s2
            startLoop:
                ; verificam daca al = valoarea din s2
                cmp al, byte[s2+ebx]
                jne okay 
                    xor edx, edx ; gasit
                okay:
                ; continui pana la capatul s2
                inc ebx
                cmp ebx, l2
                jl startLoop
            cmp edx, 1 ; nu l-am gasit in s2 => bagam in d
            jne final
            mov [D+edi], al
            inc edi
            final:
            inc esi ; urmatorul din s1
        loop Repeta
        Sfarsit:
        push dword 0
        call [exit]
